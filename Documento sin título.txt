  <script>
    // --- REFERENCIAS DOM ---
    const resEl = document.getElementById("resolution");
    const fpsEl = document.getElementById("fps");
    const codecEl = document.getElementById("codec");
    const qosEl = document.getElementById("qos");
    const trafficEl = document.getElementById("otherTraffic");
    const trafficValEl = document.getElementById("otherTrafficVal");
    const linkEl = document.getElementById("routerLink");
    const bitrateReadoutEl = document.getElementById("bitrateReadout");


    const camPersonsEl = document.getElementById("camPersons");
    const clientPersonsEl = document.getElementById("clientPersons");
    const clientDelayEl = document.getElementById("clientDelay");
    const routerQueueEl = document.getElementById("routerQueue");


    const camCanvas = document.getElementById("camCanvas");
    const clientCanvas = document.getElementById("clientCanvas");
    const netCanvas = document.getElementById("netCanvas");


    const camCtx = camCanvas.getContext("2d");
    const clientCtx = clientCanvas.getContext("2d");
    const netCtx = netCanvas.getContext("2d");


    // --- ESTADO DE LA SIMULACIÓN ---
    const state = {
      res: "1280x720",
      fps: 30,
      codec: "H.264",
      qos: "AF41",
      otherTrafficPct: 40,
      linkMbps: 20,
      videoBitrateMbps: 4.5,
      otherBitrateMbps: 8.0,
      queueMs: 0,
      totalDelayMs: 112,
      videoLossRate: 0,
      otherLossRate: 0
    };


    // Entidades de la escena
    const entities = [
      { id: 1, type: "persona", x: 185, speed: 18, baseConf: 0.83 },
      { id: 2, type: "perro",   x: 245, speed: 24, baseConf: 0.94 },
      { id: 3, type: "persona", x: 350, speed: 18, baseConf: 0.81 },
      { id: 4, type: "auto",    x: 375, speed: 32, baseConf: 0.89 }
    ];


    // Buffer de historial para simular el retardo real en el cliente
    const frameHistory = [];


    // Partículas de red y partículas descartadas
    const packets = [];
    const droppedParticles = [];
    let spawnAccumulatorVideo = 0;
    let spawnAccumulatorOther = 0;


    // --- CÁLCULO DE MÉTRICAS DE RED Y QOS ---
    function updateNetworkMetrics() {
      state.res = resEl.value;
      state.fps = parseInt(fpsEl.value, 10);
      state.codec = codecEl.value;
      state.qos = qosEl.value;
      state.otherTrafficPct = parseInt(trafficEl.value, 10);
      state.linkMbps = parseInt(linkEl.value, 10);


      trafficValEl.textContent = state.otherTrafficPct;
      trafficEl.style.background = `linear-gradient(to right, #177867 0%, #177867 ${state.otherTrafficPct}%, #ffffff ${state.otherTrafficPct}%, #ffffff 100%)`;


      const resFactor = {
        "640x480": 1.2,
        "1280x720": 3.0,
        "1920x1080": 6.5,
        "3840x2160": 18.0
      }[state.res];


      const fpsFactor = state.fps / 30;
      const codecFactor = {
        "H.265": 0.65,
        "H.264": 1.0,
        "MJPEG": 3.4
      }[state.codec];


      state.videoBitrateMbps = +(resFactor * fpsFactor * codecFactor).toFixed(2);
      state.otherBitrateMbps = +((state.otherTrafficPct / 100) * state.linkMbps * 0.95).toFixed(2);


      const totalDemand = state.videoBitrateMbps + state.otherBitrateMbps;
      const utilization = totalDemand / state.linkMbps;


      // Retardo base (inferencia YOLOv8n ~35ms + codificación + propagación red ~70ms)
      const baseProcessingDelay = 95 + (state.res === "3840x2160" ? 28 : state.res === "1920x1080" ? 12 : 5);


      if (utilization <= 0.85) {
        state.queueMs = 0;
        state.videoLossRate = 0;
        state.otherLossRate = 0;
      } else {
        const overload = utilization - 0.85;
        if (state.qos === "EF") {
          // Prioridad estricta LLQ: el video casi no sufre cola ni pérdida
          state.queueMs = Math.min(25, Math.round(overload * 35));
          state.videoLossRate = utilization > 1.35 ? Math.min(0.15, (utilization - 1.35) * 0.3) : 0;
          state.otherLossRate = Math.min(0.85, overload * 1.1);
        } else if (state.qos === "AF41") {
          // Cola reservada CBWFQ: protege el video moderadamente
          state.queueMs = Math.min(180, Math.round(overload * 190));
          state.videoLossRate = utilization > 1.05 ? Math.min(0.45, (utilization - 1.05) * 0.7) : 0;
          state.otherLossRate = Math.min(0.75, overload * 0.95);
        } else {
          // Best Effort (FIFO): video y datos compiten por igual
          state.queueMs = Math.min(450, Math.round(overload * 420));
          state.videoLossRate = utilization > 0.95 ? Math.min(0.7, (utilization - 0.95) * 0.9) : 0;
          state.otherLossRate = state.videoLossRate;
        }
      }


      state.totalDelayMs = baseProcessingDelay + state.queueMs;


      routerQueueEl.textContent = `cola del router: ${state.queueMs} ms`;
      routerQueueEl.style.color = state.queueMs > 80 ? "#b82921" : "#5a605d";


      bitrateReadoutEl.innerHTML = `
        <span>Flujo video: <strong>${state.videoBitrateMbps} Mbps</strong></span>
        <span>Otro tráfico: <strong>${state.otherBitrateMbps} Mbps</strong></span>
        <span>Uso enlace: <strong>${Math.round(utilization * 100)}%</strong></span>
        <span>Pérdida video: <strong>${Math.round(state.videoLossRate * 100)}%</strong></span>
      `;
    }


    [resEl, fpsEl, codecEl, qosEl, trafficEl, linkEl].forEach(el => {
      el.addEventListener("input", updateNetworkMetrics);
      el.addEventListener("change", updateNetworkMetrics);
    });


    // --- RESPONSIVE CANVAS ---
    function resizeCanvases() {
      [camCanvas, clientCanvas, netCanvas].forEach(c => {
        const rect = c.parentElement.getBoundingClientRect();
        const dpr = window.devicePixelRatio || 1;
        c.width = rect.width * dpr;
        c.height = rect.height * dpr;
        const ctx = c.getContext("2d");
        ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
      });
    }
    window.addEventListener("resize", resizeCanvases);


    // --- DIBUJO DE ESCENA Y DETECCIONES YOLO ---
    function drawScene(ctx, width, height, snapshot, isClient) {
      ctx.clearRect(0, 0, width, height);


      const roadY = height - 34;
      // Carretera
      ctx.fillStyle = "#d2d6d4";
      ctx.fillRect(0, roadY, width, 34);


      // Línea discontinua
      ctx.strokeStyle = "#5d6562";
      ctx.lineWidth = 2;
      ctx.setLineDash([12, 10]);
      ctx.beginPath();
      ctx.moveTo(0, roadY + 17);
      ctx.lineTo(width, roadY + 17);
      ctx.stroke();
      ctx.setLineDash([]);


      let visiblePersons = 0;
      const scaleX = width / 420;


      snapshot.forEach(item => {
        const drawX = item.x * scaleX;
        if (drawX < -50 || drawX > width + 30) return;


        if (item.type === "persona" && drawX > 0 && drawX < width - 5) {
          visiblePersons++;
        }


        // Si el cliente tiene pérdida alta de paquetes, ocasionalmente omite caja o muestra artefacto
        const packetDroppedFrame = isClient && snapshot.dropped;


        ctx.fillStyle = "#525c58";
        let boxX, boxY, boxW, boxH;


        if (item.type === "persona") {
          const headR = 7.5;
          const bodyW = 16;
          const bodyH = 44;
          const baseY = roadY - 2;
          const headY = baseY - bodyH - headR - 3;


          // Cabeza
          ctx.beginPath();
          ctx.arc(drawX + bodyW / 2, headY, headR, 0, Math.PI * 2);
          ctx.fill();


          // Cuerpo
          roundRect(ctx, drawX, baseY - bodyH, bodyW, bodyH, 3);
          ctx.fill();


          boxX = drawX - 6;
          boxY = headY - headR - 4;
          boxW = bodyW + 12;
          boxH = baseY - boxY + 2;
        } else if (item.type === "perro") {
          const baseY = roadY + 10;
          const bodyW = 28;
          const bodyH = 12;


          // Cuerpo ovalado
          ctx.beginPath();
          ctx.ellipse(drawX + bodyW / 2, baseY - bodyH / 2, bodyW / 2, bodyH / 2, 0, 0, Math.PI * 2);
          ctx.fill();


          // Cabeza del perro
          ctx.beginPath();
          ctx.arc(drawX + bodyW - 3, baseY - bodyH - 2, 4.5, 0, Math.PI * 2);
          ctx.fill();


          boxX = drawX - 3;
          boxY = baseY - bodyH - 8;
          boxW = bodyW + 8;
          boxH = bodyH + 10;
        } else if (item.type === "auto") {
          const baseY = roadY + 8;
          const carW = 54;
          const carH = 26;


          // Chasis
          roundRect(ctx, drawX, baseY - carH, carW, carH - 4, 5);
          ctx.fill();
          // Cabina superior
          roundRect(ctx, drawX + 10, baseY - carH - 13, carW - 16, 15, 4);
          ctx.fill();
          // Rueda
          ctx.fillStyle = "#363d3a";
          ctx.beginPath();
          ctx.arc(drawX + 14, baseY - 3, 6, 0, Math.PI * 2);
          ctx.arc(drawX + carW - 14, baseY - 3, 6, 0, Math.PI * 2);
          ctx.fill();


          boxX = drawX - 3;
          boxY = baseY - carH - 15;
          boxW = carW + 6;
          boxH = carH + 17;
        }


        // Dibujar Bounding Box YOLOv8n
        if (!packetDroppedFrame) {
          ctx.strokeStyle = "#c9680c";
          ctx.lineWidth = 2;
          ctx.strokeRect(boxX, boxY, boxW, boxH);


          const label = `${item.type} ${item.conf.toFixed(2)}`;
          ctx.font = "bold 11px ui-monospace, SFMono-Regular, Menlo, monospace";
          const textW = ctx.measureText(label).width;
